import { useEffect, useState } from "react";
import api from "../lib/api";

function Checkout() {
  const [cart, setCart] = useState(null);
  const [payment, setPayment] = useState("mpesa");
  const [loading, setLoading] = useState(true);
  const [placing, setPlacing] = useState(false);

  const [form, setForm] = useState({
    full_name: "",
    phone: "",
    region: "",
    city: "",
    street_address: "",
    notes: "",
  });

  const fetchCart = async () => {
    try {
      setLoading(true);
      const res = await api.get("/cart");
      setCart(res.data.cart);
    } catch (err) {
      console.log(err.response?.data || err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchCart();
  }, []);

  const handleChange = (e) => {
    setForm({
      ...form,
      [e.target.name]: e.target.value,
    });
  };

  const handleCheckout = async (e) => {
    e.preventDefault();

    try {
      setPlacing(true);

      const res = await api.post("/checkout", {
        ...form,
        payment_method: payment,
      });

      alert(res.data.message || "Order created successfully");

      if (res.data.order_number) {
        window.location.href = `/account?order=${res.data.order_number}`;
      } else {
        window.location.href = "/account";
      }
    } catch (err) {
      console.log(err.response?.data || err);
      alert(err.response?.data?.message || "Failed to place order");
    } finally {
      setPlacing(false);
    }
  };

  if (loading) {
    return <p className="text-center py-10">Loading checkout...</p>;
  }

  if (!cart || !cart.items || cart.items.length === 0) {
    return (
      <section className="bg-gray-50 py-10 px-4 min-h-screen">
        <div className="max-w-4xl mx-auto bg-white rounded-2xl p-8 shadow-sm text-center">
          <h2 className="text-2xl font-bold text-gray-800">No items for checkout</h2>
          <p className="text-gray-500 mt-2">Cart yako iko tupu.</p>
          <a
            href="/allproduct"
            className="inline-block mt-5 bg-green-600 text-white px-6 py-3 rounded-xl hover:bg-green-700"
          >
            Continue Shopping
          </a>
        </div>
      </section>
    );
  }

  return (
    <section className="bg-gray-50 py-10 mt-16 px-4 min-h-screen">
      <div className="max-w-7xl mx-auto grid lg:grid-cols-3 gap-8">

        <div className="lg:col-span-2 flex flex-col gap-6">

          <form onSubmit={handleCheckout} className="flex flex-col gap-6">

            <div className="bg-white rounded-2xl p-5 shadow-sm">
              <h2 className="text-xl font-semibold mb-4">Delivery Information</h2>

              <div className="grid md:grid-cols-2 gap-4">
                <input
                  name="full_name"
                  value={form.full_name}
                  onChange={handleChange}
                  className="border rounded-xl px-4 py-3"
                  placeholder="Full Name"
                  required
                />
                <input
                  name="phone"
                  value={form.phone}
                  onChange={handleChange}
                  className="border rounded-xl px-4 py-3"
                  placeholder="Phone Number"
                  required
                />
                <input
                  name="street_address"
                  value={form.street_address}
                  onChange={handleChange}
                  className="border rounded-xl px-4 py-3 md:col-span-2"
                  placeholder="Address"
                  required
                />
                <input
                  name="city"
                  value={form.city}
                  onChange={handleChange}
                  className="border rounded-xl px-4 py-3"
                  placeholder="City"
                  required
                />
                <input
                  name="region"
                  value={form.region}
                  onChange={handleChange}
                  className="border rounded-xl px-4 py-3"
                  placeholder="Region"
                  required
                />
                <textarea
                  name="notes"
                  value={form.notes}
                  onChange={handleChange}
                  className="border rounded-xl px-4 py-3 md:col-span-2"
                  placeholder="Order Notes (optional)"
                />
              </div>
            </div>

            <div className="bg-white rounded-2xl p-5 shadow-sm">
              <h2 className="text-xl font-semibold mb-4">Payment Method</h2>

              <div className="flex flex-col gap-3">
                <label className={`border rounded-xl p-4 flex items-center gap-3 cursor-pointer ${payment === "mpesa" ? "border-green-600 bg-green-50" : ""}`}>
                  <input
                    type="radio"
                    checked={payment === "mpesa"}
                    onChange={() => setPayment("mpesa")}
                  />
                  <span className="font-medium">M-Pesa</span>
                </label>

                <label className={`border rounded-xl p-4 flex items-center gap-3 cursor-pointer ${payment === "mixx" ? "border-green-600 bg-green-50" : ""}`}>
                  <input
                    type="radio"
                    checked={payment === "mixx"}
                    onChange={() => setPayment("mixx")}
                  />
                  <span className="font-medium">Mixx by Yas</span>
                </label>

                <label className={`border rounded-xl p-4 flex items-center gap-3 cursor-pointer ${payment === "cash" ? "border-green-600 bg-green-50" : ""}`}>
                  <input
                    type="radio"
                    checked={payment === "cash"}
                    onChange={() => setPayment("cash")}
                  />
                  <span className="font-medium">Cash on Delivery</span>
                </label>
              </div>

              {payment !== "cash" && (
                <div className="mt-4 bg-gray-50 p-4 rounded-xl text-sm text-gray-600">
                  Hakikisha namba yako iko sahihi ili upokee payment request.
                </div>
              )}
            </div>

            <button
              type="submit"
              disabled={placing}
              className="lg:hidden w-full bg-green-600 text-white py-3 rounded-xl font-semibold hover:bg-green-700 transition"
            >
              {placing ? "Processing..." : "Confirm Order"}
            </button>
          </form>
        </div>

        <div className="bg-white rounded-2xl p-5 shadow-sm h-fit sticky top-24">
          <h3 className="text-xl font-semibold mb-4">Order Summary</h3>

          {cart.items.map((item) => (
            <div key={item.id} className="flex gap-3 mb-4">
              <div className="w-20 aspect-square rounded-lg overflow-hidden bg-gray-100">
                <img
                  src={item.product.thumbnail}
                  className="w-full h-full object-cover"
                  alt={item.product.name}
                />
              </div>

              <div className="flex-1">
                <h4 className="text-sm font-medium">{item.product.name}</h4>
                <p className="text-xs text-gray-500">Qty: {item.quantity}</p>
              </div>

              <div className="text-sm font-semibold">
                TZS {Number(item.subtotal).toLocaleString()}
              </div>
            </div>
          ))}

          <div className="border-t pt-3 text-sm">
            <div className="flex justify-between mb-2">
              <span>Subtotal</span>
              <span>TZS {Number(cart.subtotal).toLocaleString()}</span>
            </div>

            <div className="flex justify-between mb-2">
              <span>Delivery</span>
              <span className="text-green-600">
                {Number(cart.delivery_fee) === 0 ? "Free" : `TZS ${Number(cart.delivery_fee).toLocaleString()}`}
              </span>
            </div>

            <div className="flex justify-between font-bold text-lg mt-3">
              <span>Total</span>
              <span className="text-green-600">
                TZS {Number(cart.total).toLocaleString()}
              </span>
            </div>
          </div>

          <button
            onClick={handleCheckout}
            disabled={placing}
            className="hidden lg:block w-full mt-5 bg-green-600 text-white py-3 rounded-xl font-semibold hover:bg-green-700 shadow hover:shadow-lg transition disabled:opacity-60"
          >
            {placing ? "Processing..." : "Confirm Order"}
          </button>

          <p className="text-xs text-gray-500 mt-3 text-center">
            🔒 Malipo yako yako salama
          </p>
        </div>
      </div>
    </section>
  );
}

export default Checkout;

function PageLoader() {
  return (
    <div className="min-h-screen bg-gray-50 flex items-center justify-center px-4">

        <div className="h-10 w-48 bg-gray-200 rounded-xl animate-pulse"></div>

}

  <Suspense fallback={<PageLoader />}>
